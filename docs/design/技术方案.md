```rust
/// 场景节点类型
#[derive(Debug, Clone)]
pub enum Node {
    Group(GroupNode),      // 组合节点
    Shape(ShapeNode),      // 图形节点
    Text(TextNode),        // 文本节点
    Image(ImageNode),      // 图片节点
    Path(PathNode),        // 路径节点
}

/// 场景图
pub struct SceneGraph {
    root: NodeId,
    nodes: SlotMap<NodeId, Node>,      // 节点存储
    transforms: HashMap<NodeId, Transform>,  // 变换矩阵
    bounds: HashMap<NodeId, BoundingBox>,   // 包围盒缓存
    spatial_index: QuadTree,           // 空间索引
    dirty_flags: HashSet<NodeId>,      // 脏标记
}

impl SceneGraph {
    /// 添加节点
    pub fn add_node(&mut self, parent: NodeId, node: Node) -> NodeId {
        let id = self.nodes.insert(node);
        self.set_parent(id, parent);
        self.mark_dirty(id);
        id
    }
    
    /// 移除节点
    pub fn remove_node(&mut self, id: NodeId) {
        self.remove_from_spatial_index(id);
        self.nodes.remove(id);
    }
    
    /// 查询视口内的节点
    pub fn query_viewport(&self, viewport: &Rect) -> Vec<NodeId> {
        self.spatial_index.query(viewport)
    }
    
    /// 更新变换矩阵（级联）
    pub fn update_transforms(&mut self) {
        // 深度优先遍历，更新所有变换
        self.update_transforms_recursive(self.root, Mat3::identity());
    }
}

/// 变换矩阵
#[derive(Debug, Clone, Copy)]
pub struct Transform {
    pub translation: Vec2,
    pub rotation: f32,
    pub scale: Vec2,
}

impl Transform {
    pub fn to_matrix(&self) -> Mat3 {
        Mat3::from_translation(self.translation)
            * Mat3::from_rotation(self.rotation)
            * Mat3::from_scale(self.scale)
    }
}
```

### 1.2 空间索引系统 (Spatial Index)

**目标**: 快速查询视口内的元素

```rust
// src/engine/spatial/quadtree.rs

/// 四叉树实现
pub struct QuadTree {
    root: Box<QuadNode>,
    max_depth: u32,
    max_elements_per_node: usize,
}

struct QuadNode {
    bounds: Rect,
    elements: Vec<(NodeId, BoundingBox)>,
    children: Option<[Box<QuadNode>; 4]>,  // NW, NE, SW, SE
}

impl QuadTree {
    /// 插入元素
    pub fn insert(&mut self, id: NodeId, bounds: BoundingBox) {
        self.insert_recursive(&mut self.root, id, bounds, 0);
    }
    
    /// 查询矩形区域内的所有元素
    pub fn query(&self, rect: &Rect) -> Vec<NodeId> {
        let mut result = Vec::new();
        self.query_recursive(&self.root, rect, &mut result);
        result
    }
    
    /// 移除元素
    pub fn remove(&mut self, id: NodeId) {
        self.remove_recursive(&mut self.root, id);
    }
    
    fn insert_recursive(
        &mut self,
        node: &mut QuadNode,
        id: NodeId,
        bounds: BoundingBox,
        depth: u32
    ) {
        // 如果没有子节点且元素数未超限，直接插入
        if node.children.is_none() 
            && node.elements.len() < self.max_elements_per_node 
        {
            node.elements.push((id, bounds));
            return;
        }
        
        // 否则需要细分或插入子节点
        if node.children.is_none() && depth < self.max_depth {
            self.subdivide(node);
        }
        
        // 插入到合适的子节点
        if let Some(ref mut children) = node.children {
            for child in children.iter_mut() {
                if child.bounds.intersects(&bounds.to_rect()) {
                    self.insert_recursive(child, id, bounds, depth + 1);
                }
            }
        }
    }
    
    fn query_recursive(
        &self,
        node: &QuadNode,
        rect: &Rect,
        result: &mut Vec<NodeId>
    ) {
        // 不相交，直接返回
        if !node.bounds.intersects(rect) {
            return;
        }
        
        // 收集当前节点的元素
        for (id, bounds) in &node.elements {
            if bounds.to_rect().intersects(rect) {
                result.push(*id);
            }
        }
        
        // 递归查询子节点
        if let Some(ref children) = node.children {
            for child in children.iter() {
                self.query_recursive(child, rect, result);
            }
        }
    }
}

/// 包围盒
#[derive(Debug, Clone, Copy)]
pub struct BoundingBox {
    pub min: Vec2,
    pub max: Vec2,
}

impl BoundingBox {
    pub fn from_points(points: &[Vec2]) -> Self {
        let mut min = Vec2::new(f32::INFINITY, f32::INFINITY);
        let mut max = Vec2::new(f32::NEG_INFINITY, f32::NEG_INFINITY);
        
        for p in points {
            min.x = min.x.min(p.x);
            min.y = min.y.min(p.y);
            max.x = max.x.max(p.x);
            max.y = max.y.max(p.y);
        }
        
        Self { min, max }
    }
    
    pub fn to_rect(&self) -> Rect {
        Rect::new(self.min, self.max - self.min)
    }
}
```

```rust
// src/engine/viewport/mod.rs

/// 视口管理器
pub struct Viewport {
    /// 相机位置（世界坐标）
    pub camera: Camera,
    
    /// 视口大小（屏幕坐标）
    pub screen_size: Vec2,
    
    /// 投影矩阵缓存
    projection_matrix: Mat4,
    
    /// 视图矩阵缓存
    view_matrix: Mat4,
    
    /// 脏标记
    dirty: bool,
}

#[derive(Debug, Clone)]
pub struct Camera {
    pub position: Vec2,      // 相机位置
    pub zoom: f32,           // 缩放级别 (0.1 - 10.0)
    pub rotation: f32,       // 旋转角度（可选）
}

impl Viewport {
    /// 创建新视口
    pub fn new(width: f32, height: f32) -> Self {
        Self {
            camera: Camera {
                position: Vec2::ZERO,
                zoom: 1.0,
                rotation: 0.0,
            },
            screen_size: Vec2::new(width, height),
            projection_matrix: Mat4::identity(),
            view_matrix: Mat4::identity(),
            dirty: true,
        }
    }
    
    /// 屏幕坐标 → 世界坐标
    pub fn screen_to_world(&self, screen_pos: Vec2) -> Vec2 {
        // NDC 坐标
        let ndc = Vec2::new(
            (screen_pos.x / self.screen_size.x) * 2.0 - 1.0,
            1.0 - (screen_pos.y / self.screen_size.y) * 2.0,
        );
        
        // 逆投影变换
        let inv_proj_view = (self.projection_matrix * self.view_matrix)
            .inverse()
            .unwrap();
        
        let world_pos = inv_proj_view * Vec4::new(ndc.x, ndc.y, 0.0, 1.0);
        Vec2::new(world_pos.x, world_pos.y)
    }
    
    /// 世界坐标 → 屏幕坐标
    pub fn world_to_screen(&self, world_pos: Vec2) -> Vec2 {
        let proj_view = self.projection_matrix * self.view_matrix;
        let clip_pos = proj_view * Vec4::new(world_pos.x, world_pos.y, 0.0, 1.0);
        
        Vec2::new(
            (clip_pos.x + 1.0) * 0.5 * self.screen_size.x,
            (1.0 - clip_pos.y) * 0.5 * self.screen_size.y,
        )
    }
    
    /// 获取可见区域（世界坐标）
    pub fn get_visible_bounds(&self) -> Rect {
        let min = self.screen_to_world(Vec2::ZERO);
        let max = self.screen_to_world(self.screen_size);
        Rect::from_corners(min, max)
    }
    
    /// 平移视口
    pub fn pan(&mut self, delta: Vec2) {
        self.camera.position += delta / self.camera.zoom;
        self.dirty = true;
    }
    
    /// 缩放视口
    pub fn zoom(&mut self, delta: f32, focal_point: Vec2) {
        let old_zoom = self.camera.zoom;
        self.camera.zoom = (self.camera.zoom * delta).clamp(0.1, 10.0);
        
        // 保持焦点位置不变
        let world_focal = self.screen_to_world(focal_point);
        self.camera.position += world_focal * (old_zoom - self.camera.zoom);
        
        self.dirty = true;
    }
    
    /// 更新矩阵
    pub fn update_matrices(&mut self) {
        if !self.dirty {
            return;
        }
        
        // 正交投影矩阵
        let half_width = self.screen_size.x * 0.5;
        let half_height = self.screen_size.y * 0.5;
        self.projection_matrix = Mat4::orthographic_rh(
            -half_width, half_width,
            -half_height, half_height,
            -1.0, 1.0,
        );
        
        // 视图矩阵
        self.view_matrix = Mat4::from_scale_rotation_translation(
            Vec3::new(self.camera.zoom, self.camera.zoom, 1.0),
            Quat::from_rotation_z(self.camera.rotation),
            Vec3::new(-self.camera.position.x, -self.camera.position.y, 0.0),
        );
        
        self.dirty = false;
    }
}
```

```rust
// src/engine/renderer/mod.rs

use wgpu::*;

/// 渲染器
pub struct Renderer {
    device: Device,
    queue: Queue,
    surface: Surface,
    config: SurfaceConfiguration,
    
    // 渲染管线
    shape_pipeline: ShapePipeline,
    text_pipeline: TextPipeline,
    image_pipeline: ImagePipeline,
    
    // 批量渲染器
    batch_renderer: BatchRenderer,
    
    // 资源管理
    texture_cache: TextureCache,
    font_atlas: FontAtlas,
}

impl Renderer {
    /// 初始化渲染器
    pub async fn new(canvas: &web_sys::HtmlCanvasElement) -> Result<Self> {
        // 1. 请求 GPU 适配器
        let instance = wgpu::Instance::new(wgpu::InstanceDescriptor {
            backends: wgpu::Backends::BROWSER_WEBGPU,
            ..Default::default()
        });
        
        let surface = instance.create_surface_from_canvas(canvas)?;
        
        let adapter = instance
            .request_adapter(&wgpu::RequestAdapterOptions {
                power_preference: wgpu::PowerPreference::HighPerformance,
                compatible_surface: Some(&surface),
                force_fallback_adapter: false,
            })
            .await
            .ok_or("Failed to request adapter")?;
        
        // 2. 请求设备和队列
        let (device, queue) = adapter
            .request_device(
                &wgpu::DeviceDescriptor {
                    label: Some("Main Device"),
                    features: wgpu::Features::empty(),
                    limits: wgpu::Limits::default(),
                },
                None,
            )
            .await?;
        
        // 3. 配置表面
        let config = wgpu::SurfaceConfiguration {
            usage: wgpu::TextureUsages::RENDER_ATTACHMENT,
            format: surface.get_preferred_format(&adapter).unwrap(),
            width: canvas.width(),
            height: canvas.height(),
            present_mode: wgpu::PresentMode::Fifo,
        };
        surface.configure(&device, &config);
        
        // 4. 创建渲染管线
        let shape_pipeline = ShapePipeline::new(&device, &config);
        let text_pipeline = TextPipeline::new(&device, &config);
        let image_pipeline = ImagePipeline::new(&device, &config);
        
        Ok(Self {
            device,
            queue,
            surface,
            config,
            shape_pipeline,
            text_pipeline,
            image_pipeline,
            batch_renderer: BatchRenderer::new(),
            texture_cache: TextureCache::new(),
            font_atlas: FontAtlas::new(),
        })
    }
    
    /// 渲染一帧
    pub fn render(
        &mut self,
        scene: &SceneGraph,
        viewport: &Viewport,
    ) -> Result<()> {
        // 1. 获取当前帧
        let frame = self.surface.get_current_texture()?;
        let view = frame.texture.create_view(&TextureViewDescriptor::default());
        
        // 2. 创建命令编码器
        let mut encoder = self.device.create_command_encoder(
            &CommandEncoderDescriptor {
                label: Some("Main Encoder"),
            }
        );
        
        // 3. 开始渲染通道
        let mut render_pass = encoder.begin_render_pass(&RenderPassDescriptor {
            label: Some("Main Render Pass"),
            color_attachments: &[Some(RenderPassColorAttachment {
                view: &view,
                resolve_target: None,
                ops: Operations {
                    load: LoadOp::Clear(Color::WHITE),
                    store: true,
                },
            })],
            depth_stencil_attachment: None,
        });
        
        // 4. 查询可见元素
        let visible_bounds = viewport.get_visible_bounds();
        let visible_nodes = scene.query_viewport(&visible_bounds);
        
        // 5. 批量渲染
        self.batch_renderer.begin_batch();
        
        for node_id in visible_nodes {
            let node = scene.get_node(node_id);
            let transform = scene.get_transform(node_id);
            
            match node {
                Node::Shape(shape) => {
                    self.batch_renderer.add_shape(shape, transform);
                }
                Node::Text(text) => {
                    self.batch_renderer.add_text(text, transform);
                }
                Node::Image(image) => {
                    self.batch_renderer.add_image(image, transform);
                }
                _ => {}
            }
        }
        
        // 6. 提交批量渲染
        self.batch_renderer.flush(
            &mut render_pass,
            &self.shape_pipeline,
            &self.text_pipeline,
            &self.image_pipeline,
            viewport,
        );
        
        drop(render_pass);
        
        // 7. 提交命令
        self.queue.submit(std::iter::once(encoder.finish()));
        frame.present();
        
        Ok(())
    }
}
```

```rust
// src/engine/renderer/batch.rs

/// 批量渲染器
pub struct BatchRenderer {
    // 形状批次
    shape_batches: Vec<ShapeBatch>,
    current_shape_batch: ShapeBatch,
    
    // 文字批次
    text_batches: Vec<TextBatch>,
    current_text_batch: TextBatch,
    
    // 顶点缓冲
    vertex_buffer: Vec<Vertex>,
    index_buffer: Vec<u32>,
}

#[derive(Debug)]
struct ShapeBatch {
    vertices: Vec<Vertex>,
    indices: Vec<u32>,
    texture: Option<TextureId>,
    blend_mode: BlendMode,
}

impl BatchRenderer {
    /// 开始新批次
    pub fn begin_batch(&mut self) {
        self.shape_batches.clear();
        self.text_batches.clear();
        self.vertex_buffer.clear();
        self.index_buffer.clear();
    }
    
    /// 添加形状到批次
    pub fn add_shape(&mut self, shape: &ShapeNode, transform: &Transform) {
        let vertices = shape.tessellate(transform);
        let indices = shape.get_indices();
        
        // 检查是否可以合并到当前批次
        if self.can_merge_to_current_batch(&shape) {
            let index_offset = self.current_shape_batch.vertices.len() as u32;
            
            self.current_shape_batch.vertices.extend(vertices);
            self.current_shape_batch.indices.extend(
                indices.iter().map(|i| i + index_offset)
            );
        } else {
            // 创建新批次
            self.flush_current_shape_batch();
            self.current_shape_batch = ShapeBatch {
                vertices,
                indices,
                texture: shape.texture(),
                blend_mode: shape.blend_mode(),
            };
        }
    }
    
    /// 刷新所有批次到 GPU
    pub fn flush(
        &mut self,
        render_pass: &mut RenderPass,
        shape_pipeline: &ShapePipeline,
        text_pipeline: &TextPipeline,
        image_pipeline: &ImagePipeline,
        viewport: &Viewport,
    ) {
        // 刷新形状批次
        for batch in &self.shape_batches {
            shape_pipeline.render(render_pass, batch, viewport);
        }
        
        // 刷新文字批次
        for batch in &self.text_batches {
            text_pipeline.render(render_pass, batch, viewport);
        }
    }
}
```

```rust
pub struct InteractionManager {

/// 事件系统
pub struct EventDispatcher {
    listeners: HashMap<EventType, Vec<Box<dyn EventListener>>>,
    event_queue: VecDeque<Event>,
}

#[derive(Debug, Clone)]
pub enum Event {
    PointerDown { id: u32, position: Vec2, button: MouseButton },
    PointerMove { id: u32, position: Vec2 },
    PointerUp { id: u32, position: Vec2 },
    Wheel { delta: Vec2, position: Vec2 },
    KeyDown { key: Key, modifiers: Modifiers },
    KeyUp { key: Key },
}

impl EventDispatcher {
    /// 分发事件
    pub fn dispatch(&mut self, event: Event) {
        // 1. 添加到队列
        self.event_queue.push_back(event.clone());
        
        // 2. 调用监听器
        if let Some(listeners) = self.listeners.get(&event.event_type()) {
            for listener in listeners {
                listener.on_event(&event);
            }
        }
    }
    
    /// 注册监听器
    pub fn add_listener(&mut self, event_type: EventType, listener: Box<dyn EventListener>) {
        self.listeners
            .entry(event_type)
            .or_insert_with(Vec::new)
            .push(listener);
    }
}

/// 手势识别器
pub struct GestureRecognizer {
    active_gestures: Vec<Gesture>,
}

#[derive(Debug)]
pub enum Gesture {
```

```rust
    max_history: usize,

impl History {
    /// 执行命令
    pub fn execute(&mut self, command: Box<dyn Command>) {
        command.execute();
        self.undo_stack.push(command);
        self.redo_stack.clear();
    }
    
    /// 撤销
    pub fn undo(&mut self) -> Option<()> {
        let command = self.undo_stack.pop()?;
        command.undo();
        self.redo_stack.push(command);
        Some(())
    }
    
    /// 重做
    pub fn redo(&mut self) -> Option<()> {
        let command = self.redo_stack.pop()?;
        command.execute();
        self.undo_stack.push(command);
        Some(())
    }
}

/// 命令模式
pub trait Command {
    fn execute(&self);
    fn undo(&self);
    fn redo(&self) {
        self.execute();
    }
}
```

```rust
        y: f32,
        width: f32,
        height: f32,
        color: &str,
    ) -> String {
        let node = ShapeNode::rectangle(x, y, width, height, parse_color(color));
        let id = self.state.document.scene.add_node(ROOT_NODE, Node::Shape(node));
        id.to_string()
    }

    #[wasm_bindgen]
    pub fn remove_element(&mut self, id: &str) {
        if let Ok(node_id) = NodeId::from_str(id) {
            self.state.document.scene.remove_node(node_id);
        }
    }

        self.state.viewport.camera.zoom = zoom;
    }

        self.engine.render(&self.state.document.scene, &self.state.viewport)

    }
```
